function makeModel(data) {
  const model = new THREE.Group();
  const bones = data.skeleton.map((b) => { const o = new THREE.Bone(); o.name = b.name; return o; });
  data.skeleton.forEach((b, i) => { applyTRS(bones[i], b.rest.t, b.rest.r, b.rest.s); (b.parent < 0 ? model : bones[b.parent]).add(bones[i]); });
  const skeleton = new THREE.Skeleton(bones, data.skeleton.map((b) => new THREE.Matrix4().fromArray(b.inverse_bind)));
  const mats = data.materials.map((m) => {
    if (m.name.includes("telegraph"))                    // telegraphs: see-through, like in the game
      return new THREE.MeshBasicMaterial({ vertexColors: true, skinning: true, transparent: true, opacity: 0.62, depthWrite: false, side: THREE.DoubleSide });
    const mat = new THREE.MeshStandardMaterial({ color: 0xffffff, vertexColors: true, roughness: m.roughness, metalness: m.metallic || 0, skinning: true, side: THREE.DoubleSide });
    if (m.emission_strength > 0 && m.emission_from_vertex_color) {
      const k = Math.min(m.emission_strength, 4).toFixed(3);
      mat.onBeforeCompile = (sh) => { sh.fragmentShader = sh.fragmentShader.replace("#include <emissivemap_fragment>", `#include <emissivemap_fragment>\n  totalEmissiveRadiance = vColor * ${k};`); };
      mat.customProgramCacheKey = () => "vc-emissive-" + k;
    }
    return mat;
  });
  for (const m of data.meshes) {
    const g = new THREE.BufferGeometry();
    g.setAttribute("position", new THREE.Float32BufferAttribute(m.positions, 3));
    g.setAttribute("normal", new THREE.Float32BufferAttribute(m.normals, 3));
    const c = new Float32Array(m.vertex_count * 3);
    for (let i = 0; i < m.vertex_count; i++) { c[i * 3] = m.colors[i * 4]; c[i * 3 + 1] = m.colors[i * 4 + 1]; c[i * 3 + 2] = m.colors[i * 4 + 2]; }
    g.setAttribute("color", new THREE.BufferAttribute(c, 3));
    const idx = new Uint16Array(m.vertex_count * 4), w = new Float32Array(m.vertex_count * 4);
    m.bone.forEach((b, i) => { idx[i * 4] = b; w[i * 4] = 1; });
    g.setAttribute("skinIndex", new THREE.Uint16BufferAttribute(idx, 4));
    g.setAttribute("skinWeight", new THREE.Float32BufferAttribute(w, 4));
    const all = [], gm = [];
    for (const sm of m.submeshes) { g.addGroup(all.length, sm.indices.length, gm.length); gm.push(mats[sm.material]); for (const k of sm.indices) all.push(k); }
    g.setIndex(m.vertex_count > 65535 ? new THREE.Uint32BufferAttribute(all, 1) : new THREE.Uint16BufferAttribute(all, 1));
    const mesh = new THREE.SkinnedMesh(g, gm);
    mesh.frustumCulled = false;
    mesh.bind(skeleton, new THREE.Matrix4());
    model.add(mesh);
  }
  return { model, bones };
}

function applyTRS(b, t, r, s) { b.position.set(t[0], t[1], t[2]); b.quaternion.set(r[0], r[1], r[2], r[3]); b.scale.set(s[0], s[1], s[2]); }
function sampleCh(arr, i0, i1, a, out) {
  if (arr.length === 1) { for (let k = 0; k < arr[0].length; k++) out[k] = arr[0][k]; return; }
  const A = arr[i0], B = arr[i1];
  for (let k = 0; k < A.length; k++) out[k] = A[k] + (B[k] - A[k]) * a;
}
function pose(anim, frame, bs) {
  const i0 = Math.floor(frame), i1 = Math.min(i0 + 1, anim.frames - 1), a = frame - i0;
  for (const tr of anim.tracks) {
    const b = bs[tr.bone];
    sampleCh(tr.t, i0, i1, a, T); sampleCh(tr.s, i0, i1, a, S);
    if (tr.r.length === 1) q.fromArray(tr.r[0]); else { q.fromArray(tr.r[i0]); q2.fromArray(tr.r[i1]); q.slerp(q2, a); }
    b.position.fromArray(T); b.quaternion.copy(q); b.scale.fromArray(S);
  }
}

const clock = new THREE.Clock();
function tick() {
  const dt = Math.min(clock.getDelta(), 0.1);
  if (data && current) {
    const a = data.animations[current], last = a.frames - 1;
    if (playing) {
      frame += dt * 30 * speed;
      if (frame >= last) {
        if ($("repeat").checked) { frame = a.loop ? frame % last : 0; fired = -1; }
        else { frame = last; playing = false; $("play").textContent = "Play"; }
      }
      for (const e of a.events) if (e.frame > fired && e.frame <= frame) flash(e.name.replace(/_/g, " "));
      fired = Math.floor(frame);
    }
    pose(a, frame, model.bones);
    if (fx) pose(fx.anim, Math.min(frame, fx.anim.frames - 1), fx.bones);
    $("scrub").value = Math.round(frame);
    $("frame").textContent = `F ${String(Math.round(frame)).padStart(3, " ")} / ${last}`;
  }
  controls.update();
  renderer.render(scene, camera);
  requestAnimationFrame(tick);
}
resize();
loadModel($("model").value);
tick();
</script>
