function makeModel(data) {
  const model = new THREE.Group();
  const bones = data.skeleton.map((b) => {
    const bone = new THREE.Bone();
    bone.name = b.name;
    return bone;
  });
  const rest = data.skeleton.map((b) => ({ t: b.rest.t, r: b.rest.r, s: b.rest.s }));
  data.skeleton.forEach((b, i) => {
    applyTRS(bones[i], rest[i].t, rest[i].r, rest[i].s);
    (b.parent < 0 ? model : bones[b.parent]).add(bones[i]);
  });
  const inverses = data.skeleton.map((b) => new THREE.Matrix4().fromArray(b.inverse_bind));
  const skeleton = new THREE.Skeleton(bones, inverses);
  const hasColors = data.meshes.every((m) => m.colors);
  const mats = data.materials.map((m) => {
    if (m.emission) {                                   // format v1: material colors only
      const e = m.emission, peak = Math.max(e[0], e[1], e[2]);
      return new THREE.MeshStandardMaterial({
        color: new THREE.Color(m.base_color[0], m.base_color[1], m.base_color[2]),
        roughness: m.roughness, metalness: 0, skinning: true,
        emissive: peak > 0 ? new THREE.Color(e[0] / peak, e[1] / peak, e[2] / peak) : new THREE.Color(0),
        emissiveIntensity: peak > 0 ? Math.min(peak, 4) : 0,
      });
    }
    // format v2: per-vertex colors (already resolved to the base color where a material has none)
    const mat = new THREE.MeshStandardMaterial({
      color: 0xffffff, vertexColors: hasColors, roughness: m.roughness, metalness: m.metallic, skinning: true,
    });
    if (m.emission_strength > 0 && !m.emission_from_vertex_color) {
      mat.emissive = new THREE.Color(m.emission_color[0], m.emission_color[1], m.emission_color[2]);
      mat.emissiveIntensity = Math.min(m.emission_strength, 4);
    }
    if (m.emission_strength > 0 && m.emission_from_vertex_color) {
      const k = Math.min(m.emission_strength, 4).toFixed(3);
      mat.onBeforeCompile = (sh) => {
        sh.fragmentShader = sh.fragmentShader.replace("#include <emissivemap_fragment>",
          `#include <emissivemap_fragment>\n  totalEmissiveRadiance = vColor * ${k};`);
      };
      mat.customProgramCacheKey = () => "vc-emissive-" + k;
    }
    return mat;
  });
  for (const m of data.meshes) {
    const g = new THREE.BufferGeometry();
    g.setAttribute("position", new THREE.Float32BufferAttribute(m.positions, 3));
    g.setAttribute("normal", new THREE.Float32BufferAttribute(m.normals, 3));
    if (m.colors) {
      const c = new Float32Array(m.vertex_count * 3);
      for (let i = 0; i < m.vertex_count; i++) { c[i * 3] = m.colors[i * 4]; c[i * 3 + 1] = m.colors[i * 4 + 1]; c[i * 3 + 2] = m.colors[i * 4 + 2]; }
      g.setAttribute("color", new THREE.BufferAttribute(c, 3));
    }
    const idx = new Uint16Array(m.vertex_count * 4), w = new Float32Array(m.vertex_count * 4);
    m.bone.forEach((b, i) => { idx[i * 4] = b; w[i * 4] = 1; });   // one bone per vertex, weight 1
    g.setAttribute("skinIndex", new THREE.Uint16BufferAttribute(idx, 4));
    g.setAttribute("skinWeight", new THREE.Float32BufferAttribute(w, 4));
    const all = [];
    const groupMats = [];
    for (const sm of m.submeshes) {
      g.addGroup(all.length, sm.indices.length, groupMats.length);
      groupMats.push(mats[sm.material]);
      for (const k of sm.indices) all.push(k);
    }
    g.setIndex(all.length > 65535 || m.vertex_count > 65535 ? new THREE.Uint32BufferAttribute(all, 1) : new THREE.Uint16BufferAttribute(all, 1));
    const mesh = new THREE.SkinnedMesh(g, groupMats);
    mesh.frustumCulled = false;
    mesh.bind(skeleton, new THREE.Matrix4());
    model.add(mesh);
  }
  return { model, bones, mats };
}
function build(data) {
  if (model) scene.remove(model);
  let mats;
  ({ model, bones, mats } = makeModel(data));
  bossMats = mats;
  clipped = false;
  scene.add(model);
  let lo = [1e9, 1e9, 1e9], hi = [-1e9, -1e9, -1e9];
  for (const m of data.meshes) for (let i = 0; i < m.positions.length; i += 3)
    for (let k = 0; k < 3; k++) { lo[k] = Math.min(lo[k], m.positions[i + k]); hi[k] = Math.max(hi[k], m.positions[i + k]); }
  const size = Math.max(hi[0] - lo[0], hi[1] - lo[1], hi[2] - lo[2]);
  const cy = (lo[1] + hi[1]) / 2;
  controls.target.set((lo[0] + hi[0]) / 2, cy, (lo[2] + hi[2]) / 2);
  camera.position.set(controls.target.x + size * 0.75, cy + size * 0.35, controls.target.z + size * 1.45);
  ring.visible = grid.visible = size > 5;
  controls.update();
}

const q = new THREE.Quaternion(), q2 = new THREE.Quaternion();
function applyTRS(bone, t, r, s) {
  bone.position.set(t[0], t[1], t[2]);
  bone.quaternion.set(r[0], r[1], r[2], r[3]);
  bone.scale.set(s[0], s[1], s[2]);
}
function sampleCh(arr, i0, i1, a, out) {
  if (arr.length === 1) { const v = arr[0]; for (let k = 0; k < v.length; k++) out[k] = v[k]; return out; }
  const A = arr[i0], B = arr[i1];
  for (let k = 0; k < A.length; k++) out[k] = A[k] + (B[k] - A[k]) * a;
  return out;
}
const T = [0, 0, 0], R = [0, 0, 0, 1], S = [1, 1, 1];
function pose(anim, frame, bs = bones) {
  const i0 = Math.floor(frame), i1 = Math.min(i0 + 1, anim.frames - 1), a = frame - i0;
  anim.tracks.forEach((tr) => {
    const b = bs[tr.bone];
    sampleCh(tr.t, i0, i1, a, T); sampleCh(tr.s, i0, i1, a, S);
    if (tr.r.length === 1) q.fromArray(tr.r[0]);
    else { q.fromArray(tr.r[i0]); q2.fromArray(tr.r[i1]); q.slerp(q2, a); }
    b.position.fromArray(T); b.quaternion.copy(q); b.scale.fromArray(S);
  });
}
function snapshot() {
  return bones.map((b) => ({ p: b.position.clone(), q: b.quaternion.clone(), s: b.scale.clone() }));
}
function applyBlend(from, a) {
  bones.forEach((b, i) => {
    const f = from[i];
    b.position.lerpVectors(f.p, b.position, a);
    b.quaternion.copy(f.q.clone().slerp(b.quaternion, a));
    b.scale.lerpVectors(f.s, b.scale, a);
  });
}

// ---- boss spawn: the black hole plays alongside, everything below the floor is clipped
const FLOOR = [new THREE.Plane(new THREE.Vector3(0, 1, 0), 0)];
let bossMats = [], fx = null, fxLoading = false, clipped = false;
function isSpell() { return !!(data && data.animations.loop && data.animations.vanish); }
function setClip(on) {
  if (on === clipped) return;
  clipped = on;
  for (const m of bossMats) { m.clippingPlanes = on ? FLOOR : []; m.needsUpdate = true; }
}
function ensureFx() {
  if (fx || fxLoading) return;
  fxLoading = true;
  getJson("spell-black-hole.json").catch(() => getJson("../export/spell-black-hole.json")).then((j) => {
    const built = makeModel(j);
    fx = { data: j, model: built.model, bones: built.bones };
    fx.model.visible = false;
    scene.add(fx.model);
  }).catch(() => { fxLoading = false; });
}
function updateFx(anim) {
  const on = !isSpell() && current === "spawn";
  setClip(on);
  if (!fx) { if (on) ensureFx(); return; }
  fx.model.visible = on;
  if (!on) return;
  const A = fx.data.animations, sp = A.spawn.frames - 1, lp = A.loop.frames - 1, vn = A.vanish.frames - 1;
  const done = (anim.events.find((e) => e.name === "spawn_done") || { frame: anim.frames - 1 }).frame;
  if (frame < sp) pose(A.spawn, frame, fx.bones);
  else if (frame < done) pose(A.loop, (frame - sp) % lp, fx.bones);
  else pose(A.vanish, Math.min(frame - done, vn), fx.bones);
}
function select(name, fromUser, keepQueue) {
  if (name === "__sequence") { queue = SEQUENCE.slice(1); name = SEQUENCE[0]; keepQueue = true; }
  if (!keepQueue) queue = [];
  const anim = data.animations[name];
  if (!anim) return;
  if (current && xfade && fromUser) { blendFrom = snapshot(); blendT = 0; }
  current = name; frame = 0; firedUpTo = -1; playing = true;
  $("play").textContent = "Pause";
  repeat = $("repeat").checked;
  document.querySelectorAll(".anim").forEach((el) => el.classList.toggle("active", el.dataset.name === (keepQueue ? "__sequence" : name)));
  $("now").textContent = name.replace(/_/g, " ");
  $("sub").textContent = `${anim.frames} Frames · ${anim.duration.toFixed(2)} s · ${anim.loop ? "loop" : "once"}`;
  const scrub = $("scrub");
  scrub.max = anim.frames - 1;
  const marks = $("marks");
  marks.innerHTML = "";
  for (const e of anim.events) {
    const m = document.createElement("div");
    m.className = "mark";
    m.style.left = `${(e.frame / (anim.frames - 1)) * 100}%`;
    m.innerHTML = `<span>${e.name} · F${e.frame}</span>`;
    marks.append(m);
  }
}
function flash(text) {
  const f = $("flash");
  f.textContent = text;
  f.classList.add("on");
  requestAnimationFrame(() => requestAnimationFrame(() => f.classList.remove("on")));
}

function buildList() {
  const list = $("list");
  list.innerHTML = "";
  const known = new Set();
  const seqOk = SEQUENCE.every((n) => data.animations[n]);
  const groups = GROUPS.map(([title, items]) => [title, items.filter(([n]) =>
    (title === "Spell") === isSpell() && (data.animations[n] || (n === "__sequence" && seqOk)))]);
  const extra = Object.keys(data.animations).filter((n) => !GROUPS.some(([, it]) => it.some(([m]) => m === n)));
  if (extra.length) groups.push(["More", extra.map((n) => [n, ""])]);
  for (const [title, items] of groups) {
    if (!items.length) continue;
    const g = document.createElement("div");
    g.className = "group";
    g.innerHTML = `<h2>${title}</h2>`;
    const l = document.createElement("div");
    l.className = "list";
    for (const [name, desc] of items) {
      const a = data.animations[name] || { events: [], loop: false,
        frames: SEQUENCE.reduce((s, n) => s + data.animations[n].frames, 0) };
      const b = document.createElement("button");
      b.className = "anim";
      b.dataset.name = name;
      const ev = a.events.length ? ` <span class="ev">· ${a.events.map((e) => e.name).join(", ")}</span>` : "";
      b.innerHTML = `<span class="name">${name === "__sequence" ? "Sequence" : name.replace(/_/g, " ")}${a.loop ? '<span class="pill">Loop</span>' : ""}</span>` +
        `<span class="len">${a.frames} F</span><span class="desc">${desc}${ev}</span>`;
      b.addEventListener("click", () => select(name, true));
      l.append(b);
      known.add(name);
    }
    g.append(l);
    list.append(g);
  }
}

const seqOk = () => SEQUENCE.every((n) => data.animations[n]);

  data = json;
  build(data);
  buildList();
  const names = Object.keys(data.animations);
  const verts = data.meshes.reduce((s, m) => s + m.vertex_count, 0);
  $("meta").textContent = `${data.skeleton.length} Bones · ${verts.toLocaleString("en-US")} Vertices · ${names.length} animations · ${data.fps} fps · ${data.coordinate_system.up} up`;
  $("status").hidden = true;
  select(data.animations.idle ? "idle" : data.animations.Idle ? "Idle" : seqOk() ? "__sequence" : names[0], false);
}

$("play").addEventListener("click", () => {
  if (!current) return;
  const anim = data.animations[current];

function step(d) {
  if (!current) return;
  playing = false; $("play").textContent = "Play";
  const anim = data.animations[current];
  frame = Math.max(0, Math.min(anim.frames - 1, Math.round(frame) + d));
}

const clock = new THREE.Clock();
function tick() {
  const dt = Math.min(clock.getDelta(), 0.1);
  if (data && current) {
    const anim = data.animations[current];
    const last = anim.frames - 1;
    if (playing) {
      frame += dt * data.fps * speed;
      if (frame >= last) {
        if (queue.length) { select(queue.shift(), false, true); requestAnimationFrame(tick); return; }
        if (repeat && current && data.animations.spawn && $("list").querySelector(".active")?.dataset.name === "__sequence") {
          select("__sequence", false); requestAnimationFrame(tick); return;
        }
        if (repeat) { frame = anim.loop ? frame % last : 0; firedUpTo = -1; }
        else { frame = last; playing = false; $("play").textContent = "Play"; }
      }
      for (const e of anim.events) {
        if (e.frame > firedUpTo && e.frame <= frame) flash(`${e.name} · F${e.frame}`);
      }
      firedUpTo = Math.floor(frame);
    }
    pose(anim, frame);
    updateFx(anim);
    if (blendFrom && blendT < 1) {
      blendT = Math.min(1, blendT + dt / 0.15);
      applyBlend(blendFrom, blendT * blendT * (3 - 2 * blendT));
    }
    $("scrub").value = Math.round(frame);
    $("frame").textContent = `F ${String(Math.round(frame)).padStart(3, " ")} / ${last}`;
  }
  controls.update();
  renderer.render(scene, camera);
  requestAnimationFrame(tick);
}
