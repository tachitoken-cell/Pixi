  const MAT = { solid: new THREE.MeshLambertMaterial({ vertexColors: true }), glow: new THREE.MeshBasicMaterial({ vertexColors: true, toneMapped: false }) };
  const _q = new THREE.Quaternion(), _m = new THREE.Matrix4(), UP = new THREE.Vector3(0, 1, 0);
  function meshGeo(m, idx) {
    const g = new THREE.BufferGeometry(), c = new Float32Array(m.vertex_count * 3);
    for (let i = 0; i < m.vertex_count; i++) { c[i * 3] = m.colors[i * 4]; c[i * 3 + 1] = m.colors[i * 4 + 1]; c[i * 3 + 2] = m.colors[i * 4 + 2]; }
    g.setAttribute("position", new THREE.Float32BufferAttribute(m.positions, 3)); g.setAttribute("normal", new THREE.Float32BufferAttribute(m.normals, 3));
    g.setAttribute("color", new THREE.BufferAttribute(c, 3)); g.setIndex(idx); return g;
  }
  function build(json) {
    const bones = json.skeleton.map((b) => { const o = new THREE.Object3D(); o.position.fromArray(b.rest.t); o.quaternion.set(...b.rest.r); o.scale.fromArray(b.rest.s); return o; });
    const root = new THREE.Group();
    json.skeleton.forEach((b, i) => (b.parent >= 0 ? bones[b.parent] : root).add(bones[i]));
    for (const m of json.meshes) for (const sm of m.submeshes) {
      const glow = json.materials[sm.material].emission_from_vertex_color, mesh = new THREE.Mesh(meshGeo(m, sm.indices), glow ? MAT.glow : MAT.solid);
      mesh.castShadow = !glow; mesh.receiveShadow = !glow; bones[m.bone[0]].add(mesh);
    }
    const inst = { root, t: 0, clip: null, once: null };
    inst.play = (n, once) => { inst.clip = json.animations[n] ? n : Object.keys(json.animations)[0]; inst.t = 0; inst.once = once || null; };
    inst.update = (dt) => {
      const a = json.animations[inst.clip]; if (!a) return; inst.t += dt; let f = inst.t * (json.fps || 24);
      if (a.loop) f %= Math.max(1, a.frames - 1); else if (f >= a.frames - 1) { f = a.frames - 1; if (inst.once) { const cb = inst.once; inst.once = null; cb(); } }
      const i0 = Math.floor(f), i1 = Math.min(i0 + 1, a.frames - 1), u = f - i0;
      a.tracks.forEach((tr, bi) => {
        const o = bones[bi];
        if (tr.t.length === 1) o.position.fromArray(tr.t[0]); else { const p = tr.t[i0], q = tr.t[i1]; o.position.set(lerp(p[0], q[0], u), lerp(p[1], q[1], u), lerp(p[2], q[2], u)); }
        if (tr.r.length === 1) o.quaternion.set(...tr.r[0]); else { o.quaternion.set(...tr.r[i0]); _q.set(...tr.r[i1]); o.quaternion.slerp(_q, u); }
        if (tr.s.length === 1) o.scale.fromArray(tr.s[0]); else { const p = tr.s[i0], q = tr.s[i1]; o.scale.set(lerp(p[0], q[0], u), lerp(p[1], q[1], u), lerp(p[2], q[2], u)); }
      });
    };
    inst.play(Object.keys(json.animations)[0]);
    return inst;
  }
  function staticGeo(json) {
    const inst = build(json); inst.root.updateMatrixWorld(true);
    const parts = { solid: [], glow: [] };
    inst.root.traverse((o) => { if (o.isMesh) { const g = o.geometry.toNonIndexed(); g.applyMatrix4(o.matrixWorld); parts[o.material === MAT.glow ? "glow" : "solid"].push(g); } });
    const merge = (gs) => { if (!gs.length) return null; const n = gs.reduce((s, g) => s + g.attributes.position.count, 0);
      const P = new Float32Array(n * 3), N = new Float32Array(n * 3), C = new Float32Array(n * 3); let o = 0;
      for (const g of gs) { P.set(g.attributes.position.array, o * 3); N.set(g.attributes.normal.array, o * 3); C.set(g.attributes.color.array, o * 3); o += g.attributes.position.count; }
      const m = new THREE.BufferGeometry(); m.setAttribute("position", new THREE.BufferAttribute(P, 3)); m.setAttribute("normal", new THREE.BufferAttribute(N, 3)); m.setAttribute("color", new THREE.BufferAttribute(C, 3)); return m; };
    return { solid: merge(parts.solid), glow: merge(parts.glow) };
  }

  // ------------------------------------------------ one map
  let MAP = null, ST = null, animated = [], mobs = [], labels = [];
  const hAt = (x, z) => { const X = Math.floor(x), Z = Math.floor(z); if (!MAP || X < 0 || Z < 0 || X >= MAP.size[0] || Z >= MAP.size[1]) return null; const c = MAP.s[Z * MAP.size[0] + X]; return c === " " ? null : MAP.h[Z * MAP.size[0] + X]; };
  const sAt = (x, z) => MAP.s[Math.floor(z) * MAP.size[0] + Math.floor(x)];
  function terrain(map, st) {
    const n = map.size[0], pos = [], nor = [], cl = [];
    const inter = st.kind === "interior";
    const pal = (k) => (Array.isArray(k) ? k.map(col) : col(k));
    const C = inter ? { floor: pal(st.floor), tile: pal(st.tile), wall: col(st.wall), wtop: col(st.wall_top), line: col(st.line) }
      : { top: pal(st.top), path: col(st.path), shore: col(st.shore), water: col(st.water), side: col(st.side), cobble: st.cobble ? pal(st.cobble) : null, bridge: col("#8a6a44") };
    const colorOf = (x, z, c) => {
      if (inter) return c === "W" ? C.wtop : c === "l" ? C.line : c === "t" ? C.tile[(x + z) % 2] : C.floor[Math.floor(hash(x, z, 2) * 2)];
      return c === "p" ? C.path : c === "c" ? C.cobble[Math.floor(hash(x, z, 5) * 4)] : c === "b" ? C.bridge : c === "w" ? C.water : c === "s" ? C.shore : C.top[Math.floor(hash(x, z, 1) * 3)];
    };
    const quad = (a, b, c2, d, nn, K) => { for (const v of [a, b, c2, a, c2, d]) { pos.push(...v); nor.push(...nn); cl.push(K.r, K.g, K.b); } };
    for (let z = 0; z < n; z++) for (let x = 0; x < n; x++) {
      const c = map.s[z * n + x]; if (c === " ") continue;
      let h = map.h[z * n + x]; const y = c === "w" ? h + 0.35 : h;
      const K = colorOf(x, z, c).clone().multiplyScalar(1 + (hash(z, x) - 0.5) * 0.08);
      quad([x, y, z], [x, y, z + 1], [x + 1, y, z + 1], [x + 1, y, z], [0, 1, 0], K);
      const side = inter ? (c === "W" ? C.wall : K.clone().multiplyScalar(0.7)) : K.clone().multiplyScalar(0.72);
      for (const [dx, dz, nn] of [[1, 0, [1, 0, 0]], [-1, 0, [-1, 0, 0]], [0, 1, [0, 0, 1]], [0, -1, [0, 0, -1]]]) {
        const X = x + dx, Z = z + dz; let nh;
        if (X < 0 || Z < 0 || X >= n || Z >= n || map.s[Z * n + X] === " ") nh = inter ? -1 : -2; else nh = map.s[Z * n + X] === "w" ? map.h[Z * n + X] + 0.35 : map.h[Z * n + X];
        if (nh >= y) continue;
        if (dx) { const Xe = dx > 0 ? x + 1 : x; quad([Xe, nh, z + (dx > 0 ? 1 : 0)], [Xe, nh, z + (dx > 0 ? 0 : 1)], [Xe, y, z + (dx > 0 ? 0 : 1)], [Xe, y, z + (dx > 0 ? 1 : 0)], nn, side); }
        else { const Ze = dz > 0 ? z + 1 : z; quad([x + (dz > 0 ? 0 : 1), nh, Ze], [x + (dz > 0 ? 1 : 0), nh, Ze], [x + (dz > 0 ? 1 : 0), y, Ze], [x + (dz > 0 ? 0 : 1), y, Ze], nn, side); }
      }
    }
    const g = new THREE.BufferGeometry(); g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3)); g.setAttribute("normal", new THREE.Float32BufferAttribute(nor, 3)); g.setAttribute("color", new THREE.Float32BufferAttribute(cl, 3));
    const m = new THREE.Mesh(g, MAT.solid); m.receiveShadow = true; return m;
  }
  const ANIM_KINDS = new Set(["landmark", "marker", "dungeon"]);
  let openTok = 0;
  async function openMap(id, arrive, preloaded) {
    if (!renderer) initGL();
    const tok = ++openTok;                                    // only the newest request may build the map
    $("atlas").hidden = true; $("mapview").hidden = false; $("loading").hidden = false;
    history.replaceState(null, "", `#${id}`);
    const map = preloaded || await fetch(MAPS[id].file).then((r) => r.json());
    const st = W.styles[map.style];
    const names = new Set([...map.props.map((p) => p[0]), ...map.doors.map((d) => d.asset), W.gate_asset[map.style]]);
    await Promise.all([...names].map(load));
    if (tok !== openTok) return;
    $("labels").innerHTML = "";
    if (mapGroup) scene.remove(mapGroup);
    mapGroup = new THREE.Group(); scene.add(mapGroup); animated = []; mobs = []; labels = [];
    MAP = map; ST = st;
    mapGroup.add(terrain(map, st));
    // props: animated ones as full models, the rest instanced
    const groups = {};
    for (const p of map.props) (groups[p[0]] ??= []).push(p);
    for (const [name, list] of Object.entries(groups)) {
      const json = await load(name), kind = W.assets[name]?.kind;
      if (ANIM_KINDS.has(kind) || list.length < 3 && /flame|lantern|lamp|brazier|candle|tank|orb|seal|fountain|totem/.test(name)) {
        for (const p of list) { const a = build(json); a.root.position.set(p[1], p[2], p[3]); a.root.rotation.y = p[4]; a.root.scale.setScalar(p[5]); mapGroup.add(a.root); animated.push(a); }
      } else {
        const g = (staticCache[name] ??= staticGeo(json));
        for (const k of ["solid", "glow"]) { if (!g[k]) continue;
          const im = new THREE.InstancedMesh(g[k], MAT[k], list.length);
          list.forEach((p, i) => { _m.compose(new THREE.Vector3(p[1], p[2], p[3]), new THREE.Quaternion().setFromAxisAngle(UP, p[4]), new THREE.Vector3(p[5], p[5], p[5])); im.setMatrixAt(i, _m); });
          im.castShadow = k === "solid"; im.receiveShadow = k === "solid"; mapGroup.add(im); }
      }
    }
    for (const gt of map.gates) { const a = build(await load(W.gate_asset[map.style])); a.root.position.set(gt.x, gt.y, gt.z); a.root.rotation.y = gt.ry; mapGroup.add(a.root); animated.push(a); gt.inst = a;
      labels.push({ el: label("gate", `→ ${gt.label}`, MAPS[gt.to]?.levels && MAPS[gt.to].levels !== "—" ? `Lv ${MAPS[gt.to].levels}` : "town"), x: gt.x, y: gt.y + 4.8, z: gt.z }); }
    for (const d of map.doors) { const a = build(await load(d.asset)); a.root.position.set(d.x, d.y, d.z); a.root.rotation.y = d.ry; mapGroup.add(a.root); animated.push(a); d.inst = a;
      const [cls, t, sub] = d.kind === "realm-door" ? ["door", `Realm door · ${d.label}`, `Lv ${d.levels} · a new world`] : d.kind === "return" ? ["ret", `Leave the realm`, `back to ${d.label}`] : ["dun", "Moss gate", "SP trial dungeons"];
      labels.push({ el: label(cls, t, sub), x: d.x, y: d.y + (d.kind === "dungeon" ? 7.5 : 6.8), z: d.z }); }
    // placeholder monsters in the spawn zones
    const mcol = new THREE.Color(st.kind === "interior" ? st.wall_top : st.top[0]).lerp(new THREE.Color(0x2a1020), 0.5);
    for (const sp of map.spawns) {
      const n = sp.boss ? 1 : Math.min(sp.count, LITE ? 3 : 5);
      for (let i = 0; i < n; i++) {
        const s = sp.boss ? 2.4 : 0.8 + Math.random() * 0.5, m = new THREE.Group();
        const body = new THREE.Mesh(new THREE.BoxGeometry(1.1 * s, 0.9 * s, 1.2 * s), new THREE.MeshLambertMaterial({ color: sp.boss ? 0x5a1a22 : mcol })); body.position.y = 0.45 * s; body.castShadow = true; m.add(body);
        for (const sx of [-0.22, 0.22]) { const e = new THREE.Mesh(new THREE.BoxGeometry(0.16 * s, 0.14 * s, 0.05), new THREE.MeshBasicMaterial({ color: sp.boss ? 0xff4030 : new THREE.Color(st.particles.color) })); e.position.set(sx * s, 0.62 * s, 0.62 * s); m.add(e); }
        const a = Math.random() * Math.PI * 2, r = Math.random() * sp.r; m.position.set(sp.x + Math.cos(a) * r, 0, sp.z + Math.sin(a) * r);
        m.userData = { sp, a: Math.random() * 6, spd: sp.boss ? 0.6 : 0.8 + Math.random(), s }; mapGroup.add(m); mobs.push(m);
      }
      labels.push({ el: label(sp.boss ? "mob boss" : "mob", sp.boss ? `Boss · Lv ${sp.level[0]}-${sp.level[1]}` : `Lv ${sp.level[0]}-${sp.level[1]}`), follow: mobs[mobs.length - 1], y: sp.boss ? 3.6 : 1.9 });
    }
    // look
    scene.background = new THREE.Color(st.kind === "interior" ? st.void : st.sky.horizon);
    scene.fog.color.set(st.fog.color); scene.fog.near = st.fog.near; scene.fog.far = st.fog.far + (st.kind === "interior" ? 0 : 60);
    sun.color.set(st.sun.color); sun.intensity = st.sun.intensity; hemi.color.set(st.ambient); hemi.groundColor.set(st.ambient).multiplyScalar(0.45);
    hemi.intensity = st.kind === "interior" ? 0.95 : 0.75;
    // player
    const at = arrive || map.start; S.x = at[0]; S.z = at[1]; S.target = null; armed = false;
    const g0 = map.gates.find((g) => Math.hypot(g.arrive[0] - S.x, g.arrive[1] - S.z) < 0.5);
    S.camYaw = g0 ? g0.ry : Math.PI; player.position.set(S.x, hAt(S.x, S.z) ?? 0, S.z);
    // hud
    const reg = REG[map.region];
    $("hudEb").textContent = reg.id === "overworld" ? "Overworld" : `Realm world · ${reg.name.split(" · ")[0]}`;
    $("hudName").textContent = map.name;
    $("hudSub").textContent = `${map.type} · ${map.levels === "—" ? "safe town" : "Lv " + map.levels}${map.monster_kinds ? " · " + map.monster_kinds : ""}`;
    $("miniImg").src = MAPS[id].thumb;
    $("loading").hidden = true;
    banner(reg.id === "overworld" ? "Overworld" : reg.name.split(" · ")[0], map.name, map.levels === "—" ? "Safe town" : `Lv ${map.levels}`);
    if (!running) { running = true; clock.getDelta(); requestAnimationFrame(frame); }
  }
  function label(cls, text, sub) { const el = document.createElement("span"); el.className = cls; el.innerHTML = text + (sub ? `<small>${sub}</small>` : ""); $("labels").append(el); return el; }
  function banner(eb, t, sub) { const B = $("banner"); B.querySelector(".eb").textContent = eb; B.querySelector("h3").textContent = t; B.querySelector("p").textContent = sub; B.classList.add("on"); clearTimeout(banner.t); banner.t = setTimeout(() => B.classList.remove("on"), 2600); }
  function toast(t) { const T = $("toast"); T.textContent = t; T.classList.add("on"); clearTimeout(toast.t); toast.t = setTimeout(() => T.classList.remove("on"), 2600); }
  // one trip at a time: lock, play the portal, load the next map once, place the player, unlock
  let travelling = false, armed = true;
  const wait = (ms) => new Promise((r) => setTimeout(r, ms));
  async function travel(to, arriveFn, inst) {
    if (travelling) return;
    travelling = true; S.busy = true; S.target = null; $("act").hidden = true;
    if (inst) inst.play("enter", null);
    $("fade").classList.add("on");
    try {
      const [m] = await Promise.all([fetch(MAPS[to].file).then((r) => r.json()), wait(inst ? 700 : 300)]);
      await openMap(to, arriveFn(m), m);
    } catch (e) { console.error(e); toast("Could not open that map."); }
    finally { if (inst) inst.play("idle"); $("fade").classList.remove("on"); travelling = false; S.busy = false; }
  }

  // ------------------------------------------------ loop
  const clock = new THREE.Clock(), v3 = new THREE.Vector3();
  function frame() {
    if (!running) return;
    const dt = Math.min(0.05, clock.getDelta()), T = clock.elapsedTime;
    let mx = 0, mz = 0; const k = S.keys, fx = -Math.sin(S.camYaw), fz = -Math.cos(S.camYaw);
    if (k.w || k.arrowup) { mx += fx; mz += fz; } if (k.s || k.arrowdown) { mx -= fx; mz -= fz; } if (k.d || k.arrowright) { mx -= fz; mz += fx; } if (k.a || k.arrowleft) { mx += fz; mz -= fx; }
    if (S.target) { mx = S.target.x - S.x; mz = S.target.z - S.z; if (Math.hypot(mx, mz) < 0.4) { S.target = null; mx = mz = 0; } }
    const len = Math.hypot(mx, mz);
    if (len > 0.01 && !S.busy) {

    // travel: map gates are walk-in (they re-arm once you have stepped away after arriving); doors use the button / E

    }

    renderer.render(scene, camera);

  }
